    G * L = 1 →
      imPart G = Gᴴ * -imPart L * G ∧
        imPart G = G * -imPart L * Gᴴ

    fromBlocks (1 + (2 * I) • portResponse YR G YR) ((2 * I) • portResponse YR G YC)
